<cfsetting showdebugoutput="false">

<cfparam name="url.id" default="">

<!--- ID validation --->
<cfif NOT reFind("^\d+$", url.id)>
    <cfheader statuscode="400" statustext="Bad Request">
    <cfabort>
</cfif>

<!--- PDF Folder --->
<cfset pdfFolder =
    expandPath("./pdf/" & url.id & "/")>

<!--- Folder ရှိ/မရှိ --->
<cfif NOT directoryExists(pdfFolder)>
    <cfheader statuscode="404" statustext="Not Found">
    <cfabort>
</cfif>

<!--- PDF ရှာ --->
<cfdirectory
    action="list"
    directory="#pdfFolder#"
    name="pdfFiles"
    filter="*.pdf">

<!--- PDF မရှိ --->
<cfif pdfFiles.recordCount EQ 0>
    <cfheader statuscode="404" statustext="Not Found">
    <cfabort>
</cfif>

<!--- တကယ့် physical file path --->
<cfset pdfPath =
    pdfFolder & pdfFiles.name[1]>

<!--- PDF ကို browser ဆီ stream လုပ် --->
<cfcontent
    type="application/pdf"
    file="#pdfPath#"
    deletefile="false">

.............................................................................................
<cfparam
    name="form.machineId"
    default="">


<!---
    1. Machine ID validation
--->
<cfif NOT reFind("^\d+$", form.machineId)>

    <h3>機械を正しく選択してください。</h3>
    <cfabort>

</cfif>


<!---
    2. PDF root folder
--->
<cfset pdfRoot =
    expandPath("../pdf/")>


<!---
    3. Selected machine folder
--->
<cfset targetFolder =
    pdfRoot & form.machineId & "/">


<!---
    4. Folder မရှိရင် create
--->
<cfif NOT directoryExists(targetFolder)>

    <cfdirectory
        action="create"
        directory="#targetFolder#">

</cfif>


<!---
    5. PDF Upload
--->
<cftry>

    <cffile
        action="upload"
        filefield="pdfFile"
        destination="#targetFolder#"
        nameconflict="makeunique"
        accept="application/pdf">


    <!---
        Extension ထပ်စစ်မယ်
    --->
    <cfif lCase(cffile.serverFileExt) NEQ "pdf">

        <cffile
            action="delete"
            file="#cffile.serverDirectory#/#cffile.serverFile#">

        <h3>
            PDFファイルのみアップロードできます。
        </h3>

        <cfabort>

    </cfif>


    <h3>
        PDFを登録しました。
    </h3>

    <p>
        ファイル名：
        <cfoutput>
            #encodeForHTML(cffile.serverFile)#
        </cfoutput>
    </p>

    <p>
        <a href="pdfRegister.cfm">
            戻る
        </a>
    </p>


    <cfcatch>

        <h3>
            PDFの登録に失敗しました。
        </h3>

        <cfoutput>
            #encodeForHTML(cfcatch.message)#
        </cfoutput>

    </cfcatch>

</cftry>

<script>

const machineSelect =
    document.getElementById("machineSelect");

const pdfArea =
    document.getElementById("pdfArea");


machineSelect.addEventListener(
    "change",
    function () {

        const machineId = this.value;


        // ဘာမှမရွေးထားရင်
        if (!machineId) {

            pdfArea.innerHTML = "";

            return;
        }


        // Loading
        pdfArea.innerHTML =
            "<p>PDFを確認しています...</p>";


        fetch(
            "getPdf.cfm?id=" +
            encodeURIComponent(machineId)
        )

        .then(response => {

            if (!response.ok) {
                throw new Error(
                    "Server Error"
                );
            }

            return response.json();

        })

        .then(data => {

            if (data.exists) {

                pdfArea.innerHTML =
                    '<iframe ' +
                    'id="pdfViewer" ' +
                    'src="' +
                    data.url +
                    '">' +
                    '</iframe>';

            }
            else {

                pdfArea.innerHTML =
                    '<p id="noPdf">' +
                    'PDFは登録されていません。' +
                    '</p>';

            }

        })

        .catch(error => {

            console.error(error);

            pdfArea.innerHTML =
                "<p>PDFの取得に失敗しました。</p>";

        });

    }
);

</script>


</body>
</html>

<cfsetting showdebugoutput="false">

<cfparam
    name="url.id"
    default="">


<cfset result = {
    "exists" = false,
    "url" = ""
}>


<!---
    ID က number ပဲဖြစ်ခွင့်ပေးမယ်
--->
<cfif reFind("^\d+$", url.id)>


    <!---
        ဥပမာ:
        /pdf/2/
    --->
    <cfset pdfFolder =
        expandPath("./pdf/" & url.id & "/")>


    <!---
        Folder ရှိ/မရှိ
    --->
    <cfif directoryExists(pdfFolder)>


        <!---
            PDF files ရှာ
        --->
        <cfdirectory
            action="list"
            directory="#pdfFolder#"
            name="pdfFiles"
            filter="*.pdf">


        <!---
            PDF တစ်ခုခုရှိရင်
        --->
        <cfif pdfFiles.recordCount GT 0>

            <cfset fileName =
                pdfFiles.name[1]>


            <cfset result.exists = true>


            <cfset result.url =
                "./pdf/" &
                url.id &
                "/" &
                encodeForURL(fileName)>

        </cfif>

    </cfif>

</cfif>


<cfcontent
    type="application/json"
    reset="true">

<cfoutput>
#serializeJSON(result)#
</cfoutput>
